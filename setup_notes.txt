#instructions for setting up repo and environment

git clone https://github.com/icanswim/qchem.git
cd qchem

##recommended uv environment
uv sync
uv run jupyter lab

##conda environment
conda env create -f environment.yml
conda activate qchem
pip install torcheval
jupyter lab

##manual creation of conda environment and folders
conda create --name qchem python=3.12
conda activate qchem
conda config --add channels nvidia pytorch pyg conda-forge
conda install pytorch torchvision torchaudio pytorch-cuda=12.6 rdkit pyg
conda install matplotlib scipy pandas scikit-learn jupyterlab h5py git numpy
pip install torcheval
jupyter lab
